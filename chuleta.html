Imprímela: es lo único que puedes llevar a la prueba.

Mostrar y comentar

# esto es un comentario
print("Hola")

Variables

nombre = "Ana"      # texto
vidas = 3          # entero
precio = 2.5       # decimal
vidas = vidas - 1

Operadores

+  -  *  /     # suma, resta, producto, división
%              # resto (7 % 2 → 1)

Pedir datos

nombre = input("Nombre: ")
edad = int(input("Edad: "))
precio = float(input("Precio: "))

Decidir

if nota >= 9:
    print("Sobresaliente")
elif nota >= 5:
    print("Aprobado")
else:
    print("Suspenso")

Comparar

==  !=  <  >  <=  >=

Repetir

for i in range(5):        # 0,1,2,3,4
    print(i)
for i in range(1, 6):     # 1..5
    print(i)

Acumulador

suma = 0
for i in range(1, 5):
    suma = suma + i
